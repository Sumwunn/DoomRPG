Class PandRandomSpawner : RandomSpawner
{
  static int GetGameLevel()
	{
	let g = PandGlobalVariables.Get();
	if(!pan_progressivespawn)
		return 1000;
	return g.GameLevel;
	}

  bool GetBossStatus(DropItem ass)
	{
	class<PandMonster> pclass = ass.Name;
	if(pclass == null || !GetDefaultByType(pclass).bPANDBOSS) return true;
	if(!pan_pandbosses) return false;
	if(!pan_bossprogression) return true;
	let g = PandGlobalVariables.Get();
	if(GetDefaultByType(pclass).bMIDTIERBOSS && g.KilledLowTier < 3) return false;
	if(GetDefaultByType(pclass).bHIGHTIERBOSS && g.KilledMidTier < 3) return false;
	if(GetDefaultByType(pclass).bSUPERTIERBOSS && g.KilledHighTier < 3) return false;
	return true;
	}

  override void PostSpawn(Actor spawned)
	{
	let pmon = PandMonster(spawned);
	if(pmon && !pmon.spawner)
		{
		let spawnname = GetClassName();
		pmon.spawner = spawnname;
		}
	}

  override Name ChooseSpawn()
	{
	DropItem di;   // di will be our drop item list iterator
	DropItem drop; // while drop stays as the reference point.
	int n = 0;
	bool nomonsters = sv_nomonsters || level.nomonsters;
			

	drop = di = GetDropItems();
	if(di != null)
		{
		while(di != null)
			{
			let prob = di.Probability;
			let level = prob * pan_progressionslider;
			bool shouldSkip = (di.Name == 'None') || (nomonsters && IsMonster(di)) || GetGameLevel() < level;

			if (!shouldSkip)
				{
				int amt = di.Amount;
				if(amt < 0) amt = 1; // default value is -1, we need a positive value.
				n += amt; // this is how we can weight the list.
				}

			di = di.Next;
			}
			
			if (n == 0)
				{ // Nothing left to spawn. They must have all been monsters, and monsters are disabled.
				return 'None';
				}
				
			// Then we reset the iterator to the start position...
			di = drop;
			
			// Take a random number...
			n = random[randomspawn](0, n-1);
			
			// And iterate in the array up to the random number chosen.
			while (n > -1 && di != null)
			{
			let prob = di.Probability;
			let level = prob * pan_progressionslider;
			
			if (di.Name != 'None' && (!nomonsters || !IsMonster(di)) && (level <= GetGameLevel() && GetBossStatus(di)))
				{
				int amt = di.Amount;
				if (amt < 0) amt = 1;
				n -= amt;
				if ((di.Next != null) && (n > -1))
					di = di.Next;
				else
					n = -1;
				}
			else
				{
				di = di.Next;
				}
			}
			
		if (di == null)
			{
			return 'Unknown';
			}
		
		return di.Name;
		}
	else
		{
		return 'None';
		}
	}
}

  void Pand_Replace(class<actor> replclass = null, bool force = false)
	{
	if(health < 0 || (force == false && target) || (force == false && !spawner))
		return;
	actor newmobj;
	if(replclass)
		newmobj = Spawn(replclass,pos,NO_REPLACE);
	else
		newmobj = Spawn(spawner,pos,NO_REPLACE);
	bool boss = false;
	if(newmobj != null)
		{
			// copy everything relevant
			newmobj.SpawnAngle = SpawnAngle;
			newmobj.Angle		= Angle;
			newmobj.Pitch		= Pitch;
			newmobj.Roll		= Roll;
			newmobj.SpawnPoint = SpawnPoint;
			newmobj.special    = special;
			newmobj.args[0]    = args[0];
			newmobj.args[1]    = args[1];
			newmobj.args[2]    = args[2];
			newmobj.args[3]    = args[3];
			newmobj.args[4]    = args[4];
			newmobj.special1   = special1;
			newmobj.special2   = special2;
			newmobj.SpawnFlags = SpawnFlags & ~MTF_SECRET;	// MTF_SECRET needs special treatment to avoid incrementing the secret counter twice. It had already been processed for the spawner itself.
			newmobj.HandleSpawnFlags();
			newmobj.SpawnFlags = SpawnFlags;
			newmobj.bCountSecret = SpawnFlags & MTF_SECRET;	// "Transfer" count secret flag to spawned actor
			newmobj.ChangeTid(tid);
			newmobj.Vel	= Vel;
			newmobj.master = master;	// For things such as DamageMaster/DamageChildren, transfer mastery.
			newmobj.target = target;
			newmobj.tracer = tracer;
			newmobj.CopyFriendliness(self, false);
			if (!bDropped) newmobj.bDropped = false;
			// Handle special altitude flags
			if (newmobj.bSpawnCeiling)
			{
				newmobj.SetZ(newmobj.ceilingz - newmobj.Height - SpawnPoint.Z);
			}
			else if (newmobj.bSpawnFloat) 
			{
				double space = newmobj.ceilingz - newmobj.Height - newmobj.floorz;
				if (space > 48)
				{
					space -= 40;
					newmobj.SetZ((space * random[randomspawn]()) / 256. + newmobj.floorz + 40);
				}
				newmobj.AddZ(SpawnPoint.Z);
			}
			if (newmobj.bMissile && !(newmobj is 'RandomSpawner'))
				newmobj.CheckMissileSpawn(0);
			// Bouncecount is used to count how many recursions we're in.
			if (newmobj is 'RandomSpawner')
				newmobj.bouncecount = ++bouncecount;
			// If the spawned actor has either of those flags, it's a boss.
			if (newmobj.bBossDeath || newmobj.bBoss)
				boss = true;
			// If a replaced actor has either of those same flags, it's also a boss.
			readonly<Actor> rep = GetDefaultByType(GetReplacee(GetClass()));
			if (rep && (rep.bBossDeath || rep.bBoss))
				boss = true;
		}
	ClearCounters();
	if (boss)
		{
		tracer = newmobj;
		SetStateLabel("Oopsie");
		A_ChangeLinkFLags(1);
		bNOINTERACTION = true;
		bISMONSTER = false;
		for(int a = 0;a<10;a++)
			A_StopSound(a);
		}
	else
		{
		A_RemoveChildren(RMVF_EVERYTHING);
		Destroy();
		}
	}

  States
  {
  Crush:
  GenericCrush:
	TNT1 A 0
		{
		let g = VerySatisfyingSquish(Spawn("VerySatisfyingSquish",pos));
		if(g)
			{
			g.master = self;
			g.translation = bloodtranslation;
			}

	Stop;

		{
		if(!tracer || tracer.health <= 0)
			{
			A_BossDeath();
			Destroy();
			}
		}
	Wait;
  	}
}

	{
	let g = PandGlobalVariables.Get();
	string otherspec = "None";
	if(other is "PandMonster")
		otherspec = PandMonster(other).spec2;
	if(g && g.SpawnEventActive == 6 && otherspec == "Chaos") return false;
	if(other.master && other.master is "ChaosMonolith") return false;
	if(other.target && other.target.master && other.bNOBLOCKMAP == true && other.target.master is "ChaosMonolith" && other.target.Species == "Chaos") return false;
	else return Super.CanCollideWith(other,passive);
	}

Mixin Class AbyssalEnemy
{
  Default
  {
  PandMonster.Species2 "Abyssal";
  PandMonster.TagColor "[v4]";
  StencilColor "DarkBlue";
  }

  int dmgtaken;
  int dmgwait;
  int rush;

  override int DamageMobj(Actor inflictor, Actor source, int damage, Name mod, int flags, double angle)
	{
	if(source && source.bISMONSTER == true && source is "PandMonster")
		{
		let source2 = PandMonster(source);
		if(source.bDONTHARMSPECIES == true && spec2 != "None" && spec2 == source2.spec2)
			return 0;
		}
	dmgtaken += damage;
	if(!bNOPAIN && dmgtaken >= 200 && health > 0)
		{
		bNOPAIN = true;
		GiveInventory("AbyssalResistance",1);
		if(CountInv("AbyssalDashCooldown"))
			{
			A_StartSound("Abyssal/Resistance",11);
			bNOPAIN = false;
			}
		else
			{
			dmgtaken = 50;
			GiveInventory("AbyssalDashCooldown",1);
			SetStateLabel("PainTeleport");
			}
		}
	return Actor.DamageMobj(inflictor,source,damage,mod,flags,angle);
	}

  override void Tick()
	{
	Super.Tick();
	if(health > 0 && !IsFrozen())
		{
		A_SpawnItemEx("AbyssalGhost",-1,0,0,-0.5,frandom(-0.5,0.5),0.5,0,SXF_TRANSFERSCALE|SXF_TRANSFERSPRITEFRAME);
		A_SpawnParticle("0000FF",SPF_RELATIVE,random(10,14),frandom(4,8),random(0,360),random(0,radius),0,random(4,height),frandom(0,5),0,frandom(1,3),0,0,2,1,-1,-1);
		A_SpawnItemEx("AbyssalSmoke",frandom(0,radius-10),0,frandom(5,height),frandom(-0.5,0.5),0,frandom(-1,1),random(0,360));
		A_SpawnItemEx("AbyssalSmoke",frandom(0,radius-10),0,frandom(0,5),frandom(0,1.5),0,0,random(0,360));
		}
	if(dmgtaken > 0 && dmgwait >= 35)
		{
		dmgtaken -= 25;
		if(dmgtaken < 0)
			dmgtaken = 0;
		dmgwait = 0;
		}
	else
		dmgwait++;
	}

  void AbyssalChase(int chasetic)
	{
	if(random(1,80) == 1 && !rush)
		rush = 100;
	if(rush)
		{
		A_Recoil(-1);
		A_SpawnItemEx("AbyssalDashGhost",0,0,0,0,0,0,0,SXF_TRANSFERSCALE|SXF_TRANSFERSTENCILCOL|SXF_TRANSFERSPRITEFRAME);
		A_SetTics(chasetic);
		rush--;
		}
	if(CountInv("AbyssalDashCooldown"))
		A_Chase(null,null);
	else
		A_Chase();
	}

  void AbyssalDash(int chance, int type = 1)
	{
	A_Stop();
	if(random(1,100) <= chance && health > 0)
		{
		let oldspeed = speed;
		A_SetSpeed(oldspeed*2);
		A_SpawnItemEx("AbyssalDashEffect",flags:SXF_SETMASTER);
		if(type < 3)
			{
			for(int a = 0;a<=30;a++)
				{
				if(type == 1)
					A_Chase(null,null);
				else if(type == 2)
					A_Wander();
				A_SpawnItemEx("AbyssalDashGhost",flags:SXF_TRANSFERSPRITEFRAME|SXF_TRANSFERSTENCILCOL|SXF_TRANSFERSCALE);
				for(int a = 0;a<2;a++)
					A_SpawnParticle("000FA6",SPF_FULLBRIGHT|SPF_RELATIVE,random(25,30),frandom(15,25),random(0,360),frandom(0,20),0,frandom(10,100),frandom(0,2),0,frandom(-2,2),0,0,0,1,-1,-1);
				}
			}
		A_SetSpeed(oldspeed);
		A_SpawnItemEx("AbyssalDashEffect",flags:SXF_SETMASTER);
		}
	}
}
