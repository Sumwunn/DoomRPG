class Pand_ReplacerHandler : EventHandler
{
  override void CheckReplacement(ReplaceEvent a)
	{
	let g = PandGlobalVariables.Get();
	//Blood and gibs
	if(pan_blood) 
		{	
		switch(a.Replacee.GetClassName()) 
			{
			case 'Blood' 		: a.Replacement = 'PandBlood'; break;
			case 'BloodSplatter': a.Replacement = 'PandBlood'; break;
			}
		}
	else
		{
		switch(a.Replacee.GetClassName()) 
			{
			case 'PandBloodSubtract' 	: a.Replacement = 'Blood'; break;
			case 'GibGenerator' 		: a.Replacement = 'Nothing'; break;
			case 'SmallGibGenerator' 	: a.Replacement = 'Nothing'; break;
			case 'BossGibGenerator' 	: a.Replacement = 'Nothing'; break;
			case 'SubtractGibGenerator' : a.Replacement = 'Nothing'; break;
			}
		}

	//Monster presets
	if(pan_monsterpatch == 1)
		{
		switch(a.Replacee.GetClassName()) 
			{
			case 'WolfensteinSS'	: a.Replacement = 'WolfensteinSSSpawner'; break;
			case 'CommanderKeen'	: a.Replacement = 'Sayori'; break;
			}
		}
	else if(pan_monsterpatch == 2)
		{
		switch(a.Replacee.GetClassName()) 
			{
			case 'WolfensteinSS'	: a.Replacement = 'Scythe2Marine'; break;
			case 'CommanderKeen'	: a.Replacement = 'Scythe2Afrit'; break;
			}
		}
	else if(pan_monsterpatch == 3)
		{
		switch(a.Replacee.GetClassName()) 
			{
			case 'WolfensteinSS'	: a.Replacement = 'Epic2Alien'; break;
			}
		}
		
	//Name "fixes"
	switch(a.Replacee.GetClassName())
		{
		case 'Mauler'	: a.Replacement = 'PandMauler'; break;
		case 'Sentinel'	: a.Replacement = 'PandSentinel'; break;
		case 'Wraith'	: a.Replacement = 'PandWraith'; break;
		}
		
	//Zombie Progression Spawning stuff
	switch(a.Replacee.GetClassName())
		{
		case 'ZombiemanSpawnS'		:
			if(pan_progressivespawn)
				{ a.Replacement = 'ZombiemanSpawnS2'; break; }
			else
				{ a.Replacement = 'PistolZombieSpawn'; break; }
		case 'ShotgunguySpawnS'		:
			if(pan_progressivespawn)
				{ a.Replacement = 'ShotgunguySpawnS2'; break; }
			else
				{ a.Replacement = 'NewShotgunGuySpawn'; break; }
		case 'ChaingunguySpawnS'		:
			if(pan_progressivespawn)
				{ a.Replacement = 'ChaingunguySpawnS2'; break; }
			else
				{ a.Replacement = 'AssaultGunnerSpawn'; break; }
		}
	}
}

class Pand_EffectChoker : EventHandler
{
  override void WorldThingSpawned(WorldEvent e)
	{
	if(pan_effectchoker && e.thing && e.thing is "PandFX" && PandFX(e.thing).bDontChoke == false)
		e.thing.Destroy();
	}
	
  override void NetworkProcess(ConsoleEvent e)
    {
	if(e.name != "Pand_ClearEffects" || !e.isManual)
		return;
	if(e.player != net_arbitrator)
        {
		if(e.player == consoleplayer)
		Console.Printf("Only the host can activate effect choker!");
		return;
        }        
	if(!pan_effectchoker)
		{
		CVar.GetCVar("pan_effectchoker").setint(1);
		Console.Printf("Effect Choker Enabled");
		}
	else
		{
		CVar.GetCVar("pan_effectchoker").setint(0);
		Console.Printf("Effect Choker Disabled");
		}
    }
}

class Pand_MonsterCounterHandler : EventHandler
{
  override void WorldThingSpawned(WorldEvent e)
	{
	bool counts = (e.thing && e.thing.bCountKill);
	if(!pan_lostsoulcount && counts && e.thing is "NewLostSoul")
		{
		e.thing.bCountKill = false;
		Level.total_monsters--;
		}
	}
	
  override void WorldThingRevived(WorldEvent e)
	{
	if(e.thing && e.thing.bCountKill)
		{
		e.thing.bCountKill = false;
		Level.total_monsters--;
		}
	}
}

  override void WorldUnloaded(WorldEvent e)
	{
	let g = PandGlobalVariables.Get();
	let maxchance = 6;
	if(g)
		{
		if(g.GotChaosWeapon)
			maxchance = 3;
		if(Level.MapName != "OUTPOST" && g.MonstersOnly)
			g.GameLevel += 4;
		g.GameLevel++;
		if(g.GameLevel > (40 * pan_progressionslider) && random(1,maxchance) == 1)
			g.SpawnEventActive = random(1,(SpawnEvent.Size()));
		else if(g.SpawnEventActive > 0)
			g.SpawnEventActive = 0;
		if(g.GameLevel < (60 * pan_progressionslider) && (g.SpawnEventActive == 6 || g.SpawnEventActive == 3))
			g.SpawnEventActive--;
		if(g.ForceChaosEvent)
			{
			g.ForceChaosEvent = false;
			g.SpawnEventActive = 6;
			}
		}
	}

  override void WorldThingDied(WorldEvent e)
	{
	let pmon = PandMonster(e.Thing);
	if((!pmon && e.Thing.bISMONSTER && e.Thing.bBOSS) || (pmon && pmon.bBOSS && !pmon.bLOWTIERBOSS && !pmon.bMIDTIERBOSS && !pmon.bHIGHTIERBOSS))
		{
		let g = PandGlobalVariables.Get();
		if(g && g.PickedBoss == 2)
			{
			let myrkura = Actor.Spawn("MyrkuraSpawner");
			myrkura.Warp(e.Thing);
			myrkura.target = players[0].mo;
			g.PickedBoss = 0;
			}
		}

	let mons = PandMonster(e.Thing);
	if(mons && mons.SpawnHealth() > 250/* && mons.bDropSidegrade == false*/)
		{
		let monshealth = mons.SpawnHealth();
		let chance = (monshealth/150);
		if(mons.bBOSS == true)
			{
			let chance2 = (monshealth/200);
			chance = chance+chance2;
			}
		if(random(0,256) < chance)
			{
			let dropper = Actor.Spawn("RandomSidegradeDropper");
			dropper.tracer = mons;
			}
		}
	name monname;
	if(mons) monname = mons.GetClassName();
	if(mons && pan_runes && mons.bBOSS == true && monname != "NewCyberdemon" && monname != "NewSpiderMastermind")
		{
		let monshealth = mons.SpawnHealth();
		let chance = (20+(monshealth/80));
		if(random(0,256) < chance)
			{
			let dropper = Actor.Spawn("RandomSidegradeDropper");
			dropper.tracer = mons;
			}
		}
	}
	
  override void NetworkProcess(ConsoleEvent e)
    {
	if(e.name == "Pand_SetGameLevel")
		{
		let g = PandGlobalVariables.Get();
		g.GameLevel = pan_setgamelevel;
		Console.Printf("Game level: \cd%d", g.GameLevel);
		}
	if(e.name == "Pand_IncreaseGameLevel1")
		{
		let g = PandGlobalVariables.Get();
		g.GameLevel++;
		Console.Printf("Game level: \cd%d", g.GameLevel);
		}
	if(e.name == "Pand_IncreaseGameLevel")
		{
		let g = PandGlobalVariables.Get();
		g.GameLevel += pan_setgamelevel;
		Console.Printf("Game level: \cd%d", g.GameLevel);
		}
	if(e.name == "Pand_DecreaseGameLevel1")
		{
		let g = PandGlobalVariables.Get();
		g.GameLevel--;
		Console.Printf("Game level: \cd%d", g.GameLevel);
		}
	if(e.name == "Pand_DecreaseGameLevel")
		{
		let g = PandGlobalVariables.Get();
		g.GameLevel -= pan_setgamelevel;
		Console.Printf("Game level: \cd%d", g.GameLevel);
		}
	if(e.name == "Pand_GameLevel")
		{
		let g = PandGlobalVariables.Get();
		Console.Printf("Game level: \cd%d", g.GameLevel);
		}
    }
